function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

function escapeHtml(str) {
  if (!str) return '';
  return String(str).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

function calculatePerformanceMetrics() {
  const totalAnswered = performanceHistory.length;
  if (totalAnswered === 0) return { accuracy: 50, streak: 0 };
  const correctAnswers = performanceHistory.filter(p => p.correct).length;
  const accuracy = (correctAnswers / totalAnswered) * 100;
  let streak = 0;
  for (let i = performanceHistory.length - 1; i >= 0; i--) {
    if (performanceHistory[i].correct) streak++;
    else break;
  }
  return { accuracy, streak };
}

function updateDifficulty() {
  const metrics = calculatePerformanceMetrics();
  const { accuracy, streak } = metrics;
  
  if (streak === 0 && performanceHistory.length >= 3) {
    const lastThree = performanceHistory.slice(-3);
    if (lastThree.every(p => !p.correct)) {
      if (currentDifficulty === 'hard') currentDifficulty = 'medium';
      else if (currentDifficulty === 'medium') currentDifficulty = 'easy';
    }
  }
  
  if (performanceHistory.length >= 5 && accuracy < 50) {
    if (currentDifficulty === 'hard') currentDifficulty = 'medium';
    else if (currentDifficulty === 'medium') currentDifficulty = 'easy';
  }
  
  if (accuracy >= 80 && streak >= 3) {
    if (currentDifficulty === 'easy') currentDifficulty = 'medium';
    else if (currentDifficulty === 'medium') currentDifficulty = 'hard';
  }
  
  const levelSpan = document.getElementById('currentDifficultyLevel');
  if (levelSpan) levelSpan.textContent = currentDifficulty.charAt(0).toUpperCase() + currentDifficulty.slice(1);
  const container = document.getElementById('difficultyIndicator');
  container.classList.remove('difficulty-easy', 'difficulty-medium', 'difficulty-hard');
  container.classList.add(`difficulty-${currentDifficulty}`);
  
  const accSpan = document.getElementById('statusAccuracy');
  if (accSpan) accSpan.textContent = `Accuracy: ${Math.round(accuracy)}%`;
  const streakSpan = document.getElementById('statusStreak');
  if (streakSpan) streakSpan.textContent = `Streak: ${streak}`;
}

  if (lessonXp) lessonXp.textContent = earnedXP;

}

function showFeedback(isCorrect, msg) {

function recordAnswer(exerciseId, isCorrect, xpAmount, topic, responseTime) {
  if (isLocked) return;
  isLocked = true;
  
  if (currentTimer) {
    clearInterval(currentTimer);
    currentTimer = null;
  }
  
  performanceHistory.push({
    id: exerciseId,
    correct: isCorrect,
    difficulty: currentDifficulty,
    topic: topic,
    responseTime: responseTime
  });
  
  console.log(`📊 Answer recorded - Exercise ${exerciseId}, Correct: ${isCorrect}, XP: ${xpAmount}`);
  
  if (isCorrect) {
    earnedXP += xpAmount;
    correctCount++;
  }
  
  updateDifficulty();
  updateProgress();
  
  currentIndex++;
  
  setTimeout(() => {
    moveToNextQuestion();
  }, 800);
}

function moveToNextQuestion() {
  if (currentTimer) {
    clearInterval(currentTimer);
    currentTimer = null;
  }
  
  if (currentIndex >= queue.length) {
    showCompletion();
    return;
  }
  
  isLocked = false;
  currentAnswerSelected = false;
  renderExercise();
}

function fuzzyMatch(user, correct) {
    const u = (user || '').trim().toLowerCase();
    const c = (correct || '').trim().toLowerCase();
    if (u === c) return true;
    if (Math.abs(u.length - c.length) <= 2) {
        let diff = 0;
        const minLen = Math.min(u.length, c.length);
        for (let i = 0; i < minLen; i++) {
            if (u[i] !== c[i]) diff++;
        }
        diff += Math.abs(u.length - c.length);
        return diff <= Math.min(2, Math.floor(c.length * 0.25));
    }
    return false;
}

function speakText(text, btn) {
  if (!text) return;
  if (!window.speechSynthesis) {
    if (btn) btn.textContent = '🔇 Not supported';
    return;
  }
  
  const origLabel = btn ? btn.textContent : '🔊 Play Audio';
  if (btn) {
    btn.textContent = '🔊 Playing…';
    btn.disabled = true;
  }
  
  const utterance = new SpeechSynthesisUtterance(text);
  utterance.rate = 0.85;
  utterance.pitch = 1.1;
  utterance.lang = 'en-US';
  
  utterance.onend = () => {
    if (btn) {
      btn.textContent = origLabel;
      btn.disabled = false;
    }
  };
  utterance.onerror = () => {
    if (btn) {
      btn.textContent = origLabel;
      btn.disabled = false;
    }
  };
  
  window.speechSynthesis.cancel();
  window.speechSynthesis.speak(utterance);
}

function handleMCQ(btn, chosen, correct) {
    const grid = btn.closest('.mcq-grid');
    if (!grid || grid.dataset.answered === '1') return;
    if (isLocked || currentAnswerSelected) return;
    grid.dataset.answered = '1';

    const ok = chosen.toLowerCase() === correct.toLowerCase();
    const ex = queue[currentIndex];
    const responseTime = (Date.now() - questionStartTime) / 1000;

    const allBtns = grid.querySelectorAll('.mcq-btn');
    for (let i = 0; i < allBtns.length; i++) {
        allBtns[i].style.pointerEvents = 'none';
    }
    btn.classList.add(ok ? 'correct' : 'wrong');
    if (!ok) {
        const optBtns = grid.querySelectorAll('[data-option]');
        for (let i = 0; i < optBtns.length; i++) {
            if ((optBtns[i].dataset.option || '').toLowerCase() === correct.toLowerCase()) {
                optBtns[i].classList.add('correct');
            }
        }
    }
    showFeedback(ok, ok ? '✓ Correct!' : '✗ Correct answer: ' + correct);

    currentAnswerSelected = true;
    recordAnswer(ex.id, ok, ex.xp || 10, ex.topic || 'general', responseTime);
}

function handleTF(btn, selected, correct) {
  if (isLocked || currentAnswerSelected) return;
  const ok = selected === correct;
  const ex = queue[currentIndex];
  const responseTime = (Date.now() - questionStartTime) / 1000;
  
  const grid = btn.closest('.tf-grid');
  if (grid) {
    const allBtns = grid.querySelectorAll('.tf-btn');
    allBtns.forEach(b => b.style.pointerEvents = 'none');
    btn.classList.add(ok ? 'correct' : 'wrong');
    if (!ok) {
      allBtns.forEach(b => {
        if (b.textContent.includes(correct)) b.classList.add('correct');
      });
    }
  }
  
  showFeedback(ok, ok ? '✓ Correct!' : '✗ Correct answer: ' + correct);
  currentAnswerSelected = true;
  recordAnswer(ex.id, ok, ex.xp || 10, ex.topic || 'general', responseTime);
}

function handleTextAnswer() {
  if (isLocked || currentAnswerSelected) return;
  const input = document.getElementById('answerInput');
  if (!input) return;
  
  const userAnswer = input.value.trim();
  const ex = queue[currentIndex];
  const responseTime = (Date.now() - questionStartTime) / 1000;
  const correctAnswer = (ex.answer || ex.correct || '').toLowerCase();
  const ok = fuzzyMatch(userAnswer, correctAnswer);
  
  input.disabled = true;
  input.classList.add(ok ? 'correct' : 'wrong');
  
  showFeedback(ok, ok ? '✓ Correct!' : '✗ Correct answer: ' + (ex.answer || ex.correct || ''));
  
  currentAnswerSelected = true;
  recordAnswer(ex.id, ok, ex.xp || 10, ex.topic || 'general', responseTime);
}

function handleSpeakingAnswer(expectedText, btn) {
  if (isLocked || currentAnswerSelected) return;
  
  showFeedback(true, '✓ Speaking exercise complete!');
  currentAnswerSelected = true;
  const ex = queue[currentIndex];
  recordAnswer(ex.id, true, ex.xp || 10, ex.topic || 'general', (Date.now() - questionStartTime) / 1000);
}

function getTimeLimit(ex) {
  return ex.time_limit_seconds || 45;
}

function getDifficultyClass(diff) {
  switch(diff) {
    case 'easy': return 'diff-easy';
    case 'medium': return 'diff-medium';
    case 'hard': return 'diff-hard';
    default: return 'diff-medium';
  }
}

function updateExamTimerDisplay() {
  const mins = Math.floor(examTimeLeft / 60);
  const secs = examTimeLeft % 60;
  const timerSpan = document.getElementById('timerSeconds');
  if (timerSpan) timerSpan.textContent = `${mins.toString().padStart(2,'0')}:${secs.toString().padStart(2,'0')}`;
}

function startExamTimer() {
  if (examTimer) clearInterval(examTimer);
  examTimer = setInterval(() => {
    if (examTimeLeft > 0) {
      examTimeLeft--;
      updateExamTimerDisplay();
      if (examTimeLeft <= 0) {
        clearInterval(examTimer);
        alert('⏰ Time is up!');
        showCompletion();
      }
    }
  }, 1000);
}

function stopExamTimer() { 
  if (examTimer) { 
    clearInterval(examTimer); 
    examTimer = null; 
  } 
}

function startQuestionTimer(seconds, onExpire) {
  if (currentTimer) clearInterval(currentTimer);
  let timeLeftVal = seconds;
  const timerDisplay = document.getElementById('timerDisplay');
  
  currentTimer = setInterval(() => {
    if (timeLeftVal <= 0) {
      clearInterval(currentTimer);
      currentTimer = null;
      if (onExpire && !isLocked && !currentAnswerSelected) {
        onExpire();
      }
    } else {
      timeLeftVal--;
      if (timerDisplay) timerDisplay.innerHTML = `⏱️ ${timeLeftVal}s`;
      if (timeLeftVal <= 10) timerDisplay.classList.add('timer-danger');
      else timerDisplay.classList.remove('timer-danger');
    }
  }, 1000);
}

      const exCurrent = queue[currentIndex];
      const responseTime = (Date.now() - questionStartTime) / 1000;
      showFeedback(false, 'Time is up!');
      currentAnswerSelected = true;
      recordAnswer(exCurrent.id, false, exCurrent.xp || 10, exCurrent.topic || 'general', responseTime);
    }
  });
}

// Word Bank Functions
let builtWords = [];

function setupWordBank(words, correct) {
    builtWords = [];
    const bank = document.getElementById('bank-wb');
    const builder = document.getElementById('builder-wb');
    if (!bank || !builder) return;
    
    const bankWords = bank.querySelectorAll('.bank-word');
    for (let i = 0; i < bankWords.length; i++) {
        bankWords[i].addEventListener('click', function() {
            if (this.classList.contains('used')) return;
            this.classList.add('used');
            builtWords.push(this.dataset.word);
            updateBuilder(builder, builtWords, bank, correct);
        });
    }
    
    const checkBtn = document.getElementById('checkWbBtn');
    if (checkBtn) {
        checkBtn.onclick = () => checkWordBank(correct);
    }
}

function updateBuilder(builder, words, bank, correct) {
    if (words.length === 0) {
        builder.innerHTML = '<span class="builder-placeholder">Tap words below to build sentence...</span>';
        builder.classList.remove('has-words');
    } else {
        builder.classList.add('has-words');
        let html = '';
        for (let i = 0; i < words.length; i++) {
            html += '<span class="built-word" data-idx="' + i + '">' + words[i] + ' <span class="rm">✕</span></span>';
        }
        builder.innerHTML = html;
        
        const builtWordsEls = builder.querySelectorAll('.built-word');
        for (let i = 0; i < builtWordsEls.length; i++) {
            builtWordsEls[i].querySelector('.rm').addEventListener('click', function() {
                const idx = parseInt(builtWordsEls[i].dataset.idx);
                const removed = builtWords.splice(idx, 1)[0];
                const bankWords = bank.querySelectorAll('.bank-word');
                for (let j = 0; j < bankWords.length; j++) {
                    if (bankWords[j].dataset.word === removed && bankWords[j].classList.contains('used')) {
                        bankWords[j].classList.remove('used');
                        break;
                    }
                }
                updateBuilder(builder, builtWords, bank, correct);
            });
        }
    }
}

function checkWordBank(correct) {
    if (builtWords.length === 0) {
        showFeedback(false, 'Please build a sentence first');
        return;
    }
    const isArr = Array.isArray(correct);
    let ok = false;
    const ex = queue[currentIndex];

    if (isArr) {
        ok = builtWords.length === correct.length;
        if (ok) {
            for (let i = 0; i < builtWords.length; i++) {
                if (builtWords[i].toLowerCase() !== correct[i].toLowerCase()) {
                    ok = false;
                    break;
                }
            }
        }
    } else {
        ok = builtWords.join(' ').toLowerCase() === correct.toLowerCase();
    }

    showFeedback(ok, ok ? '✓ Correct!' : '✗ Correct: ' + (isArr ? correct.join(' ') : correct));

    if (ok && ex) {
        correctCount++;
        earnedXP += ex.xp;
        updateProgress();
        recordAnswer(ex.id, true, ex.xp, ex.topic || 'general', (Date.now() - questionStartTime) / 1000);
    } else if (ex) {
        recordAnswer(ex.id, false, ex.xp, ex.topic || 'general', (Date.now() - questionStartTime) / 1000);
    }
}

// Matching Functions
let matchLeft = null;
let matchPairsMatched = 0;
let matchTotalPairs = 0;

function setupMatching(pairs, ex) {
    matchLeft = null;
    matchPairsMatched = 0;
    matchTotalPairs = pairs.length;
    
    const matchItems = document.querySelectorAll('.match-item');
    const scoreSpan = document.getElementById('match-score');
    if (scoreSpan) scoreSpan.textContent = `0 / ${matchTotalPairs} matched`;
    
    for (let i = 0; i < matchItems.length; i++) {
        matchItems[i].removeEventListener('click', matchClickHandler);
        matchItems[i].addEventListener('click', matchClickHandler);
    }
    
    function matchClickHandler() {
        if (isLocked || currentAnswerSelected) return;
        
        if (this.classList.contains('matched')) return;
        
        if (this.dataset.side === 'left') {
            document.querySelectorAll('.match-item.selected').forEach(el => el.classList.remove('selected'));
            this.classList.add('selected');
            matchLeft = this;
        } else if (matchLeft && this.dataset.side === 'right') {
            const leftVal = matchLeft.dataset.val;
            const rightVal = this.dataset.val;
            const isMatch = leftVal === rightVal;
            
            if (isMatch) {
                matchLeft.classList.add('matched');
                this.classList.add('matched');
                matchLeft.classList.remove('selected');
                matchPairsMatched++;
                if (scoreSpan) scoreSpan.textContent = `${matchPairsMatched} / ${matchTotalPairs} matched`;
                
                if (matchPairsMatched === matchTotalPairs) {
                    showFeedback(true, '✓ All matched!');
                    const responseTime = (Date.now() - questionStartTime) / 1000;
                    currentAnswerSelected = true;
                    recordAnswer(ex.id, true, ex.xp || 10, ex.topic || 'general', responseTime);
                }
            } else {
                matchLeft.classList.add('flash-wrong');
                this.classList.add('flash-wrong');
                setTimeout(() => {
                    matchLeft.classList.remove('flash-wrong', 'selected');
                    this.classList.remove('flash-wrong');
                }, 400);
            }
            matchLeft = null;
        }
    }
}

function handleWritingAnswer() {
  if (isLocked || currentAnswerSelected) return;
  const textarea = document.getElementById('writingTextarea');
  if (!textarea) return;
  
  const userAnswer = textarea.value.trim();
  const ex = queue[currentIndex];
  const responseTime = (Date.now() - questionStartTime) / 1000;
  
  const sentenceCount = (userAnswer.match(/[.!?]+/g) || []).length;
  const minSentences = ex.min_sentences || 3;
  const ok = sentenceCount >= minSentences && userAnswer.length > 50;
  
  textarea.disabled = true;
  
  showFeedback(ok, ok ? `✓ Good! (${sentenceCount} sentences)` : `✗ Please write at least ${minSentences} sentences (${sentenceCount} detected)`);
  
  if (ok) {
    currentAnswerSelected = true;
    recordAnswer(ex.id, true, ex.xp || 10, ex.topic || 'general', responseTime);
  } else {
    setTimeout(() => {
      textarea.disabled = false;
      textarea.focus();
    }, 2000);
  }
}

function startLesson() {

  console.log('Total exercises available:', EXERCISES.length);

    alert('No exercises available! Please check the YAML file.');
    return;
  }
  
  // Reset all state
  queue = [];
  currentIndex = 0;
  earnedXP = 0;
  correctCount = 0;
  performanceHistory = [];
  currentDifficulty = 'medium';
  examTimeLeft = 3600;
  isLocked = false;
  currentAnswerSelected = false;
  
  // Pick 40 random questions from ALL exercises
  let available = [...EXERCISES];
  shuffle(available);
  queue = available.slice(0, Math.min(40, available.length));
  
  console.log('📋 Exam started with', queue.length, 'questions');
  console.log('📋 Question types:', queue.map(q => q.type).join(', '));
  
  if (queue.length === 0) {

  }

  updateDifficulty();

  document.querySelector('.lesson-nav').style.display = 'none';
  document.getElementById('lesson').style.display = 'block';

  startExamTimer();
  enterFullscreen();

}
